cmake_minimum_required(VERSION 3.22)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(androforge_core LANGUAGES CXX)
endif()

add_library(androforge_core STATIC src/patch_engine.cpp src/deployment.cpp)
target_compile_features(androforge_core PUBLIC cxx_std_17)
target_include_directories(androforge_core PUBLIC include PRIVATE generated)
set_target_properties(androforge_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(androforge_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(androforge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NOT ANDROID)
    option(ANDROFORGE_BUILD_TESTS "Build portable patch engine tests" ON)
    if(ANDROFORGE_BUILD_TESTS)
        enable_testing()
        add_executable(androforge_core_tests tests/patch_engine_test.cpp)
        target_link_libraries(androforge_core_tests PRIVATE androforge_core)
        add_test(NAME androforge_core_tests COMMAND androforge_core_tests)
        add_executable(androforge_deployment_tests tests/deployment_test.cpp)
        target_link_libraries(androforge_deployment_tests PRIVATE androforge_core)
        if(MSVC)
            target_compile_options(androforge_deployment_tests PRIVATE /utf-8)
        endif()
        add_test(NAME androforge_deployment_tests COMMAND androforge_deployment_tests)
        add_executable(androforge_official_examples_tests tests/official_examples_test.cpp)
        target_link_libraries(androforge_official_examples_tests PRIVATE androforge_core)
        if(MSVC)
            target_compile_options(androforge_official_examples_tests PRIVATE /utf-8)
        endif()
        add_test(NAME androforge_official_examples_tests
                 COMMAND androforge_official_examples_tests
                         "${CMAKE_CURRENT_SOURCE_DIR}/../resources/official/examples/patches")
    endif()
    option(ANDROFORGE_BUILD_BENCHMARK "Build host patch engine benchmark" OFF)
    if(ANDROFORGE_BUILD_BENCHMARK)
        add_executable(androforge_core_benchmark tests/patch_engine_benchmark.cpp)
        target_link_libraries(androforge_core_benchmark PRIVATE androforge_core)
    endif()
endif()
